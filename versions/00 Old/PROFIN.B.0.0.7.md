```python
"""
Database initialisation and session factory.
Uses SQLAlchemy 2.x with SQLite backend.
Alembic is used for schema migrations (configured separately).
"""

from pathlib import Path
from sqlalchemy import create_engine, event
from sqlalchemy.orm import sessionmaker, Session
from app.db.models import Base

# ---------------------------------------------------------------------------
# Path configuration
# ---------------------------------------------------------------------------
DB_DIR = Path(__file__).resolve().parents[2] / "data"
DB_DIR.mkdir(parents=True, exist_ok=True)
DB_PATH = DB_DIR / "profin.db"

DATABASE_URL = f"sqlite:///{DB_PATH}"

# ---------------------------------------------------------------------------
# Engine — enforce foreign keys on every connection (SQLite default is OFF)
# ---------------------------------------------------------------------------
engine = create_engine(
    DATABASE_URL,
    connect_args={"check_same_thread": False},
    echo=False,  # Set True for SQL debug logging
)


@event.listens_for(engine, "connect")
def _set_sqlite_pragma(dbapi_conn, _connection_record) -> None:
    """Enable FK enforcement and WAL mode for every new SQLite connection."""
    cursor = dbapi_conn.cursor()
    cursor.execute("PRAGMA foreign_keys = ON")
    cursor.execute("PRAGMA journal_mode = WAL")
    cursor.close()


# ---------------------------------------------------------------------------
# Session factory
# ---------------------------------------------------------------------------
SessionLocal: sessionmaker[Session] = sessionmaker(
    bind=engine,
    autocommit=False,
    autoflush=False,
)


def get_session() -> Session:
    """Return a new SQLAlchemy session. Caller is responsible for closing it."""
    return SessionLocal()


def init_db() -> None:
    """Create all tables if they do not exist (first-run bootstrap)."""
    Base.metadata.create_all(bind=engine)
```

```python
"""
app/ingestion/excel_parser.py
==============================
PROFIN — CaixaBank Excel Parser

Responsibility:
  Read a CaixaBank .xlsx export file (no traditional header row) and return
  a clean, typed pandas DataFrame with standardised column names ready for
  validation and import.

CaixaBank file structure (24 unnamed columns, data from row 1 onward):
  Col 0  : always NaN                     → dropped
  Col 1  : IBAN full string               → iban_raw
  Col 2  : branch code                    → dropped
  Col 3  : currency (EUR)                 → currency
  Col 4  : operation date  dd/mm/yyyy     → transaction_date
  Col 5  : value date      dd/mm/yyyy     → value_date
  Col 6  : ingreso amount  (NaN if gasto) → _ingreso_raw
  Col 7  : gasto amount    (NaN if ingr.) → _gasto_raw
  Col 8  : running balance                → running_balance
  Col 9  : always NaN                     → dropped
  Col 10 : concepto común code            → concepto_comun
  Col 11 : concepto propio code           → concepto_propio
  Col 12 : always 000000000000            → dropped
  Col 13 : bank reference                 → dropped
  Col 14 : issuer / operation description → issuer_name
  Col 15-22 : complementary details       → concept  (concatenated, stripped)
  Col 23 : NaN / extra                    → dropped

Output columns (standardised):
  transaction_date  : datetime.date
  value_date        : datetime.date | None
  amount            : float  (positive = income, negative = expense)
  running_balance   : float
  concepto_comun    : str | None
  concepto_propio   : str | None
  issuer_name       : str | None
  concept           : str | None  (concat of cols 14-22)
  raw_col14         : str | None  (col 14 exactly as received)
  currency          : str         (always 'EUR' for CaixaBank ES)
  iban_raw          : str | None
"""
from __future__ import annotations

import re
from pathlib import Path
from typing import Union

import pandas as pd


# ---------------------------------------------------------------------------
# Column indices (0-based) in the raw CaixaBank .xlsx
# ---------------------------------------------------------------------------
_COL_IBAN            = 1
_COL_CURRENCY        = 3
_COL_DATE_OP         = 4
_COL_DATE_VAL        = 5
_COL_INGRESO         = 6
_COL_GASTO           = 7
_COL_SALDO           = 8
_COL_CONCEPTO_COMUN  = 10
_COL_CONCEPTO_PROPIO = 11
_COL_DETAILS_START   = 14   # issuer + details cols 14-22
_COL_DETAILS_END     = 22   # inclusive

# Row offset: row 0 is the title banner, data starts at row 1
_DATA_START_ROW = 1

# Date format used by CaixaBank
_DATE_FORMAT = "%d/%m/%Y"


# ---------------------------------------------------------------------------
# Public API
# ---------------------------------------------------------------------------

def parse_excel(file_path: Union[str, Path]) -> pd.DataFrame:
    """
    Read a CaixaBank .xlsx export and return a clean, typed DataFrame.

    Parameters
    ----------
    file_path : str or Path
        Path to the .xlsx file.

    Returns
    -------
    pd.DataFrame with columns:
        transaction_date, value_date, amount, running_balance,
        concepto_comun, concepto_propio, issuer_name, concept,
        raw_col14, currency, iban_raw

    Raises
    ------
    FileNotFoundError
        If the file does not exist.
    ValueError
        If the file cannot be read as a valid Excel workbook.
    """
    file_path = Path(file_path)

    if not file_path.exists():
        raise FileNotFoundError(
            f"Fichero no encontrado: {file_path}"
        )

    # -- Read raw: no header, all columns as object to avoid type coercion --
    try:
        raw: pd.DataFrame = pd.read_excel(
            file_path,
            header=None,
            dtype=object,
            engine="openpyxl",
        )
    except Exception as exc:
        raise ValueError(
            f"No se puede leer el fichero Excel '{file_path.name}': {exc}"
        ) from exc

    # -- Skip title/banner row (row 0), work from row 1 onward -------------
    data = raw.iloc[_DATA_START_ROW:].reset_index(drop=True).copy()

    # -- Drop fully empty rows ---------------------------------------------
    data.dropna(how="all", inplace=True)
    data.reset_index(drop=True, inplace=True)

    if data.empty:
        raise ValueError(
            "El fichero no contiene filas de datos tras omitir la cabecera."
        )

    # -- Validate minimum column count -------------------------------------
    if data.shape[1] < 23:
        raise ValueError(
            f"El fichero tiene {data.shape[1]} columnas — "
            "se esperan al menos 23. "
            "¿Es un export de CaixaBank con el formato correcto?"
        )

    # -- Extract & clean each field ----------------------------------------
    out = pd.DataFrame()

    out["iban_raw"] = (
        data.iloc[:, _COL_IBAN]
        .astype(str)
        .str.strip()
        .replace("nan", None)
    )

    out["currency"] = (
        data.iloc[:, _COL_CURRENCY]
        .astype(str)
        .str.strip()
        .replace("nan", None)
    )

    out["transaction_date"] = _parse_dates(
        data.iloc[:, _COL_DATE_OP], "Fecha operación (col 4)"
    )
    out["value_date"] = _parse_dates(
        data.iloc[:, _COL_DATE_VAL], "Fecha valor (col 5)",
        required=False
    )

    # Unify ingreso / gasto → signed amount
    # ingreso (col 6): positive float, NaN if not an income row
    # gasto   (col 7): positive float, NaN if not an expense row
    # Rule: amount = +ingreso if ingreso not NaN, else -gasto
    ingreso = _to_numeric(data.iloc[:, _COL_INGRESO], "Ingreso (col 6)")
    gasto   = _to_numeric(data.iloc[:, _COL_GASTO],   "Gasto (col 7)")
    out["amount"] = ingreso.combine_first(-gasto)

    out["running_balance"] = _to_numeric(
        data.iloc[:, _COL_SALDO], "Saldo (col 8)"
    )

    out["concepto_comun"] = _clean_str(
        data.iloc[:, _COL_CONCEPTO_COMUN]
    )
    out["concepto_propio"] = _clean_str(
        data.iloc[:, _COL_CONCEPTO_PROPIO]
    )

    # raw_col14: col 14 exactly as received (before stripping)
    out["raw_col14"] = (
        data.iloc[:, 14]
        .astype(str)
        .str.strip()
        .replace("nan", None)
    )

    # issuer_name: extract from col 14 — remove leading "Fecha de operación: DD-MM-YYYY"
    out["issuer_name"] = out["raw_col14"].apply(_extract_issuer)

    # concept: concatenate cols 14-22 (strip, drop empty, join with space)
    concept_cols = data.iloc[
        :, _COL_DETAILS_START : _COL_DETAILS_END + 1
    ]
    out["concept"] = concept_cols.apply(
        _concat_details, axis=1
    )

    return out


# ---------------------------------------------------------------------------
# Internal helpers
# ---------------------------------------------------------------------------

def _parse_dates(
    series: pd.Series,
    field_name: str,
    required: bool = True,
) -> pd.Series:
    """
    Parse a date column using the CaixaBank dd/mm/yyyy format.
    Returns a Series of datetime.date objects (or NaT/None if not required).
    """
    parsed = pd.to_datetime(
        series.astype(str).str.strip(),
        format=_DATE_FORMAT,
        dayfirst=True,
        errors="coerce",
    )
    if required and parsed.isna().any():
        bad_count = int(parsed.isna().sum())
        raise ValueError(
            f"Campo '{field_name}': {bad_count} fila(s) con fecha "
            "no válida o formato inesperado (se esperaba dd/mm/yyyy)."
        )
    return parsed.dt.date


def _to_numeric(series: pd.Series, field_name: str) -> pd.Series:
    """
    Convert a column to float. Treats blank / 'nan' strings as NaN.
    Does NOT raise on NaN — callers decide if NaN is acceptable.
    """
    cleaned = (
        series.astype(str)
        .str.strip()
        .str.replace(",", ".", regex=False)  # handle European decimal comma
        .replace({"nan": None, "": None, "None": None})
    )
    return pd.to_numeric(cleaned, errors="coerce")


def _clean_str(series: pd.Series) -> pd.Series:
    """Strip whitespace, collapse internal spaces, return None for blank/nan."""
    return (
        series.astype(str)
        .str.strip()
        .str.replace(r"\s+", " ", regex=True)
        .replace({"nan": None, "": None, "None": None})
    )


_DATE_PREFIX_RE = re.compile(
    r"^Fecha\s+de\s+operaci[oó]n\s*:\s*\d{2}-\d{2}-\d{4}\s*",
    re.IGNORECASE,
)


def _extract_issuer(raw: object) -> str | None:
    """
    Extract the issuer name from col 14.
    CaixaBank col 14 has two patterns:
      a) "Fecha de operación: DD-MM-YYYY   ISSUER NAME"
         → strip the date prefix, return stripped issuer name
      b) Plain issuer name with no date prefix
         → return as-is (stripped)
    Returns None if blank/nan.
    """
    if not raw or str(raw).strip() in ("", "nan", "None"):
        return None
    text = str(raw).strip()
    text = _DATE_PREFIX_RE.sub("", text).strip()
    # Collapse multiple internal spaces (CaixaBank pads with spaces)
    text = re.sub(r"\s{2,}", " ", text)
    return text if text else None


def _concat_details(row: pd.Series) -> str | None:
    """
    Concatenate detail columns 14-22 into a single concept string.
    Strips each cell, drops empty/nan, joins with single space.
    Returns None if all cells are empty.
    """
    parts = []
    for val in row:
        s = str(val).strip()
        if s and s not in ("nan", "None"):
            # Collapse internal spaces
            s = re.sub(r"\s{2,}", " ", s)
            parts.append(s)
    return " | ".join(parts) if parts else None
```

## `app/ingestion/validator.py`

```python
"""
app/ingestion/validator.py
===========================
PROFIN — Import Validation Layer

Validation rules (in order):
  V-01  DataFrame is not empty.
  V-02  All required output columns are present.
  V-03  transaction_date: no null values.
  V-04  amount: no null values.
  V-05  running_balance: no null values.
  V-06  currency: all rows are 'EUR'.
  V-07  iban_raw: all rows share the same IBAN.
  V-08  amount != 0: zero-amount rows flagged as warnings (not errors).
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import List

import pandas as pd


# ---------------------------------------------------------------------------
# Custom exceptions / result types
# ---------------------------------------------------------------------------

class ValidationError(Exception):
    """Blocking validation failure. Message shown directly in GUI."""


@dataclass
class ValidationWarning:
    """Non-blocking warning surfaced to the user during import preview."""
    rule:        str
    message:     str
    row_indices: List[int] = field(default_factory=list)


@dataclass
class ValidationResult:
    """Returned by validate_dataframe() on success."""
    row_count: int
    iban:      str
    warnings:  List[ValidationWarning] = field(default_factory=list)


# ---------------------------------------------------------------------------
# Required columns
# ---------------------------------------------------------------------------
_REQUIRED_COLUMNS = [
    "transaction_date",
    "value_date",
    "amount",
    "running_balance",
    "concepto_comun",
    "concepto_propio",
    "issuer_name",
    "concept",
    "raw_col14",
    "currency",
    "iban_raw",
]


# ---------------------------------------------------------------------------
# Public API
# ---------------------------------------------------------------------------

def validate_dataframe(df: pd.DataFrame) -> ValidationResult:
    """
    Run all validation rules on a parsed DataFrame.

    Returns
    -------
    ValidationResult on success.

    Raises
    ------
    ValidationError on any blocking failure (V-01 to V-07).
    """
    warnings: List[ValidationWarning] = []

    # V-01 — Not empty
    if df.empty:
        raise ValidationError(
            "El fichero no contiene filas de datos. "
            "Comprueba que el export de CaixaBank no esté vacío."
        )

    # V-02 — Required columns present
    missing_cols = [c for c in _REQUIRED_COLUMNS if c not in df.columns]
    if missing_cols:
        raise ValidationError(
            "Columnas requeridas no encontradas tras el parseo: "
            + ", ".join(f"'{c}'" for c in missing_cols)
            + ". Puede indicar un cambio en el formato de export de CaixaBank."
        )

    # V-03 — transaction_date: no nulls
    null_dates = df["transaction_date"].isna()
    if null_dates.any():
        bad_rows = df.index[null_dates].tolist()
        raise ValidationError(
            f"'Fecha operación': {len(bad_rows)} fila(s) sin fecha válida "
            f"(filas: {bad_rows[:5]}{'...' if len(bad_rows) > 5 else ''}). "
            "Comprueba que todas las filas tengan fecha en formato dd/mm/yyyy."
        )

    # V-04 — amount: no nulls
    null_amounts = df["amount"].isna()
    if null_amounts.any():
        bad_rows = df.index[null_amounts].tolist()
        raise ValidationError(
            f"'Importe': {len(bad_rows)} fila(s) sin importe válido "
            f"(filas: {bad_rows[:5]}{'...' if len(bad_rows) > 5 else ''}). "
            "Cada fila debe tener un valor en Ingreso o Gasto."
        )

    # V-05 — running_balance: no nulls
    null_balance = df["running_balance"].isna()
    if null_balance.any():
        bad_rows = df.index[null_balance].tolist()
        raise ValidationError(
            f"'Saldo': {len(bad_rows)} fila(s) sin saldo válido "
            f"(filas: {bad_rows[:5]}{'...' if len(bad_rows) > 5 else ''}). "
            "Todas las filas deben tener un saldo numérico."
        )

    # V-06 — currency: EUR only
    non_eur = df[df["currency"] != "EUR"]
    if not non_eur.empty:
        bad_currencies = non_eur["currency"].unique().tolist()
        raise ValidationError(
            "PROFIN v1 soporta únicamente la divisa EUR. "
            f"Se encontraron: {bad_currencies} en {len(non_eur)} fila(s)."
        )

    # V-07 — iban_raw: single IBAN per file
    unique_ibans = df["iban_raw"].dropna().unique()
    if len(unique_ibans) == 0:
        raise ValidationError(
            "No se encontró ningún IBAN en el fichero. "
            "Comprueba que el export sea de CaixaBank."
        )
    if len(unique_ibans) > 1:
        raise ValidationError(
            f"El fichero contiene {len(unique_ibans)} IBANs distintos: "
            f"{unique_ibans.tolist()}. "
            "PROFIN espera un fichero de una única cuenta."
        )
    detected_iban = str(unique_ibans[0]).strip()

    # V-08 — Zero amounts: warning only
    zero_amounts = df[df["amount"] == 0.0]
    if not zero_amounts.empty:
        warnings.append(ValidationWarning(
            rule="V-08",
            message=(
                f"{len(zero_amounts)} fila(s) con importe = 0,00 €. "
                "Se importarán pero pueden indicar ajustes contables."
            ),
            row_indices=zero_amounts.index.tolist(),
        ))

    return ValidationResult(
        row_count=len(df),
        iban=detected_iban,
        warnings=warnings,
    )
```

## `app/ingestion/duplicate_detector.py`

```python
"""
app/ingestion/duplicate_detector.py
=====================================
PROFIN — Duplicate Detection Logic

Duplicate detection priority rules (from spec):
  D-1  EXACT match on (transaction_date, concepto_comun, amount,
       running_balance) → near-certain duplicate, auto-skip.
       Logged with is_duplicate_flagged=True, duplicate_resolved_by_user=True.

  D-2  Match on (transaction_date, concepto_comun, amount) but DIFFERENT
       running_balance → likely a genuine repeated transaction
       (e.g., two identical payments same day). Do NOT auto-skip.
       Imported as new, NOT flagged.

  D-3  Match on (transaction_date, concepto_comun, amount, running_balance)
       that is ambiguous per known patterns → flag is_duplicate_flagged=True,
       prompt user via GUI dialog. User resolves Yes/No.
       NOTE: In v1 with the 4-field exact match in D-1, D-3 is only triggered
       for edge cases where the match is ambiguous despite exact field match
       (e.g., same-day batch transfers). Implementing as a flag list that
       the import flow GUI resolves interactively.

  D-4  Never silently guess on ambiguous cases — always defer to user.

Output:
  DuplicateCheckResult.auto_skip    : rows to skip (D-1 certain duplicates)
  DuplicateCheckResult.needs_review : rows needing user confirmation (D-3)
  DuplicateCheckResult.clean        : rows safe to import without question
"""
from __future__ import annotations

from dataclasses import dataclass, field
from datetime import date
from typing import List, Set, Tuple

import pandas as pd
from sqlalchemy.orm import Session

from app.db.models import Transaction


# ---------------------------------------------------------------------------
# Result type
# ---------------------------------------------------------------------------

@dataclass
class DuplicateCheckResult:
    """
    Result of running duplicate detection on an incoming DataFrame.

    Attributes
    ----------
    auto_skip : List[int]
        DataFrame row indices that are certain duplicates (D-1).
        These are SKIPPED automatically — no user prompt.
    needs_review : List[int]
        DataFrame row indices that are possible/ambiguous duplicates (D-3).
        GUI must prompt the user for each.
    clean : List[int]
        DataFrame row indices that are clean — safe to import.
    skip_reasons : dict[int, str]
        Human-readable reason string for each auto_skip index.
    review_reasons : dict[int, str]
        Human-readable reason string for each needs_review index.
    """
    auto_skip:      List[int] = field(default_factory=list)
    needs_review:   List[int] = field(default_factory=list)
    clean:          List[int] = field(default_factory=list)
    skip_reasons:   dict      = field(default_factory=dict)
    review_reasons: dict      = field(default_factory=dict)

    @property
    def total_incoming(self) -> int:
        return len(self.auto_skip) + len(self.needs_review) + len(self.clean)

    @property
    def summary(self) -> str:
        return (
            f"Total: {self.total_incoming} | "
            f"Limpios: {len(self.clean)} | "
            f"Auto-omitidos: {len(self.auto_skip)} | "
            f"Revisión pendiente: {len(self.needs_review)}"
        )


# ---------------------------------------------------------------------------
# Public API
# ---------------------------------------------------------------------------

def check_duplicates(
    incoming: pd.DataFrame,
    session: Session,
    user_id: int,
    account_id: int,
) -> DuplicateCheckResult:
    """
    Check each row of `incoming` against existing transactions in the DB.

    Parameters
    ----------
    incoming : pd.DataFrame
        Output of excel_parser.parse_excel() (already validated).
    session : Session
        Active SQLAlchemy session (read-only during this call).
    user_id : int
        ID of the authenticated user.
    account_id : int
        ID of the target account.

    Returns
    -------
    DuplicateCheckResult
    """
    result = DuplicateCheckResult()

    # -- Build lookup set from existing transactions ----------------------
    # Key for D-1 exact match: (date, concepto_comun, amount, running_balance)
    # Key for D-2/D-3 partial: (date, concepto_comun, amount)
    existing_exact: Set[Tuple] = set()
    existing_partial: Set[Tuple] = set()

    existing_txns = (
        session.query(
            Transaction.transaction_date,
            Transaction.concepto_comun,
            Transaction.amount,
            Transaction.running_balance,
        )
        .filter(
            Transaction.user_id    == user_id,
            Transaction.account_id == account_id,
        )
        .all()
    )

    for txn in existing_txns:
        exact_key   = (txn.transaction_date, txn.concepto_comun,
                       txn.amount, txn.running_balance)
        partial_key = (txn.transaction_date, txn.concepto_comun, txn.amount)
        existing_exact.add(exact_key)
        existing_partial.add(partial_key)

    # -- Check each incoming row -----------------------------------------
    # Track duplicates WITHIN the incoming batch itself (intra-batch)
    # to catch duplicate rows inside the same file.
    intra_batch_exact: Set[Tuple]   = set()
    intra_batch_partial: Set[Tuple] = set()

    for idx, row in incoming.iterrows():
        txn_date    = row["transaction_date"]
        conc_comun  = row.get("concepto_comun")
        amount      = row["amount"]
        balance     = row["running_balance"]

        exact_key   = (txn_date, conc_comun, amount, balance)
        partial_key = (txn_date, conc_comun, amount)

        # D-1: Exact match against DB → auto-skip
        if exact_key in existing_exact:
            result.auto_skip.append(idx)
            result.skip_reasons[idx] = (
                f"Duplicado exacto en BD: "
                f"{txn_date} | {conc_comun} | {amount:+.2f} € | "
                f"Saldo {balance:.2f} €"
            )
            continue

        # D-1 intra-batch: exact match within this same file → auto-skip
        if exact_key in intra_batch_exact:
            result.auto_skip.append(idx)
            result.skip_reasons[idx] = (
                f"Duplicado exacto dentro del fichero: "
                f"{txn_date} | {conc_comun} | {amount:+.2f} € | "
                f"Saldo {balance:.2f} €"
            )
            continue

        # D-2: Partial match (same date + concepto + amount, diff balance)
        # → This is a genuine repeated transaction. Import as clean, no flag.
        # (We do NOT add a needs_review here per spec rule D-2.)

        # D-3: Partial intra-batch match → possible duplicate within file
        # with different balance. Flag for user review.
        if partial_key in intra_batch_partial and exact_key not in intra_batch_exact:
            result.needs_review.append(idx)
            result.review_reasons[idx] = (
                f"Posible duplicado en el fichero (mismo concepto/importe, "
                f"saldo diferente): "
                f"{txn_date} | {conc_comun} | {amount:+.2f} € | "
                f"Saldo {balance:.2f} €"
            )
            # Register in intra-batch sets before continuing
            intra_batch_exact.add(exact_key)
            intra_batch_partial.add(partial_key)
            continue

        # Clean row
        result.clean.append(idx)
        intra_batch_exact.add(exact_key)
        intra_batch_partial.add(partial_key)

    return result
```

## `app/ingestion/importer.py`

```python
"""
app/ingestion/importer.py
==========================
PROFIN — Import Orchestrator

Responsibility:
  Orchestrate the full import pipeline:
    1. Parse   : excel_parser.parse_excel()
    2. Validate: validator.validate_dataframe()
    3. Duplicate check: duplicate_detector.check_duplicates()
    4. Persist : write ImportBatch + Transaction rows to DB
    5. Integrity check: trigger balance reconciliation after import

  This module contains NO GUI logic — it returns structured results
  that the GUI (app/gui/import_flow.py) uses to drive dialogs and progress.

Key design decisions:
  - All DB writes happen inside a single transaction per batch.
    On any unhandled exception, the session is rolled back cleanly.
  - User-resolved duplicate decisions are passed in as a dict
    {row_index: bool} where True = "import anyway", False = "skip".
  - TransactionType is derived from amount sign:
      amount > 0 → income
      amount < 0 → expense
      amount == 0 → unknown
  - source_file stores only the filename, not the full path (privacy).
"""
from __future__ import annotations

import logging
from dataclasses import dataclass, field
from datetime import datetime, timezone
from pathlib import Path
from typing import Dict, List, Optional

import pandas as pd
from sqlalchemy.orm import Session

logger = logging.getLogger(__name__)

# ---------------------------------------------------------------------------
# Public API — Stage 2: Commit (DB writes)
# ---------------------------------------------------------------------------

def commit_import(
    preview:           ImportPreview,
    session:           Session,
    user_id:           int,
    account_id:        int,
    user_decisions:    Dict[int, bool],
) -> ImportSummary:
    """
    Write the approved rows to the DB as a single atomic transaction.

    Parameters
    ----------
    preview        : ImportPreview from prepare_import().
    session        : Active SQLAlchemy session (will be committed here).
    user_id        : Authenticated user ID.
    account_id     : Target account ID.
    user_decisions : {row_index: True/False}
                     True  = user says "import this possible duplicate"
                     False = user says "skip this possible duplicate"
                     Rows in needs_review that are NOT in this dict are
                     skipped by default (conservative).

    Returns
    -------
    ImportSummary
    """
    df               = preview.df
    dup_result       = preview.duplicate_result
    file_path        = preview.file_path
    warnings: List[str] = [w.message for w in preview.validation_result.warnings]

    rows_imported = 0
    rows_skipped  = 0
    rows_errors   = 0

    try:
        # -- Create ImportBatch record ------------------------------------
        batch = ImportBatch(
            user_id          = user_id,
            filename         = file_path.name,
            import_timestamp = datetime.now(timezone.utc),
            rows_imported    = 0,
            rows_skipped     = 0,
            status           = ImportStatus.pending,
        )
        session.add(batch)
        session.flush()  # get batch.id before inserting transactions

        # -- Determine which rows to import ------------------------------
        # auto_skip → always skip
        # needs_review → import only if user_decisions[idx] is True
        # clean → always import
        skip_indices = set(dup_result.auto_skip)
        for idx in dup_result.needs_review:
            if not user_decisions.get(idx, False):
                skip_indices.add(idx)

        import_indices = [
            i for i in range(len(df))
            if i not in skip_indices
        ]

        # -- Insert transactions -----------------------------------------
        for idx in import_indices:
            row = df.iloc[idx]
            try:
                txn = _row_to_transaction(
                    row        = row,
                    user_id    = user_id,
                    account_id = account_id,
                    batch_id   = batch.id,
                    source_file= file_path.name,
                    is_flagged = idx in dup_result.needs_review,
                    resolved   = idx in dup_result.needs_review,
                )
                session.add(txn)
                rows_imported += 1

                # Audit log entry
                session.add(AuditLog(
                    user_id       = user_id,
                    table_affected= "transactions",
                    record_id     = None,  # filled after flush
                    action        = AuditAction.insert,
                    old_value     = None,
                    new_value     = (
                        f"{row['transaction_date']} | "
                        f"{row.get('issuer_name','')} | "
                        f"{row['amount']:+.2f}"
                    ),
                    timestamp     = datetime.now(timezone.utc),
                ))

            except Exception as row_exc:
                logger.warning(
                    "Error al insertar fila %d: %s", idx, row_exc
                )
                rows_errors += 1
                warnings.append(
                    f"Fila {idx}: error al insertar — {row_exc}"
                )

        rows_skipped = len(skip_indices)

        # -- Update batch counters & status ------------------------------
        batch.rows_imported = rows_imported
        batch.rows_skipped  = rows_skipped
        batch.status = (
            ImportStatus.completed
            if rows_errors == 0
            else ImportStatus.partial
        )

        session.commit()
        logger.info(
            "Import completado: batch_id=%d, importadas=%d, "
            "omitidas=%d, errores=%d",
            batch.id, rows_imported, rows_skipped, rows_errors,
        )

        return ImportSummary(
            import_batch_id = batch.id,
            rows_imported   = rows_imported,
            rows_skipped    = rows_skipped,
            rows_errors     = rows_errors,
            warnings        = warnings,
            status          = batch.status,
        )

    except Exception as exc:
        session.rollback()
        logger.error("Import abortado, rollback ejecutado: %s", exc)
        raise
```

```python
"""
app/categorization/rules_engine.py
====================================
PROFIN — Auto-Categorization Rules Engine

Responsibility:
  Apply AutoCategorizationRule entries to a set of transactions,
  assigning category_id automatically based on pattern matching.

Matching logic:
  - Fields checked (OR gate — match in EITHER field triggers the rule):
      * issuer_name
      * concept
  - Match types:
      contains : case-insensitive substring match
      exact    : case-insensitive full-string match
      regex    : Python re.search() match (case-insensitive)
  - Priority: lower number = higher priority (rule with priority=1
    beats priority=100). First match wins.
  - Conflict detection: if two rules of different categories match the
    SAME transaction at DIFFERENT priorities, the higher-priority rule
    wins silently. If two rules match at the SAME priority, a conflict
    is flagged and the user is notified (no auto-assignment).

Global vs. per-user rules:
  - AutoCategorizationRule.user_id = NULL  → global rule (all users)
  - AutoCategorizationRule.user_id = X     → rule for user X only
  - Per-user rules override global rules at the SAME priority.
  - Load order: global rules first, then user rules (user rules can
    shadow global rules at equal priority).

Output:
  RulesEngineResult.assigned   : {transaction_id: category_id}
  RulesEngineResult.conflicts  : {transaction_id: [RuleConflict, ...]}
  RulesEngineResult.unmatched  : [transaction_id, ...]
"""
from __future__ import annotations

import logging
import re
from dataclasses import dataclass, field
from typing import Dict, List, Optional

from sqlalchemy.orm import Session

from app.db.models import AutoCategorizationRule, MatchType, Transaction

logger = logging.getLogger(__name__)


# ---------------------------------------------------------------------------
# Result types
# ---------------------------------------------------------------------------

@dataclass
class RuleConflict:
    """Two rules matched the same transaction at the same priority."""
    transaction_id: int
    rule_a_id:      int
    rule_b_id:      int
    category_a_id:  int
    category_b_id:  int
    priority:       int
    field_matched:  str   # 'issuer_name' | 'concept' | 'both'
    pattern_a:      str
    pattern_b:      str


@dataclass
class RulesEngineResult:
    """
    Result of running the rules engine over a batch of transactions.

    Attributes
    ----------
    assigned  : dict {transaction_id: category_id} — auto-assigned.
    conflicts : dict {transaction_id: List[RuleConflict]} — needs user input.
    unmatched : List[transaction_id] — no rule matched.
    applied   : int — total rules applied (assigned + conflict rows).
    """
    assigned:  Dict[int, int]               = field(default_factory=dict)
    conflicts: Dict[int, List[RuleConflict]] = field(default_factory=dict)
    unmatched: List[int]                    = field(default_factory=list)

    @property
    def applied(self) -> int:
        return len(self.assigned) + len(self.conflicts)

    @property
    def summary(self) -> str:
        return (
            f"Asignadas: {len(self.assigned)} | "
            f"Conflictos: {len(self.conflicts)} | "
            f"Sin categoría: {len(self.unmatched)}"
        )


# ---------------------------------------------------------------------------
# Public API
# ---------------------------------------------------------------------------

def apply_rules(
    session:        Session,
    transaction_ids: List[int],
    user_id:        int,
) -> RulesEngineResult:
    """
    Apply all applicable rules to the given transaction IDs.

    Rules are loaded ordered by priority ASC (lowest = highest priority).
    Global rules (user_id IS NULL) are loaded first, then user-specific
    rules. At equal priority, user rules shadow global rules.

    Parameters
    ----------
    session         : Active SQLAlchemy session.
    transaction_ids : List of transaction primary keys to process.
    user_id         : Authenticated user (for loading per-user rules).

    Returns
    -------
    RulesEngineResult
    """
    result = RulesEngineResult()

    if not transaction_ids:
        return result

    # -- Load rules (global + user-specific), ordered by priority ASC ------
    rules = _load_rules(session, user_id)
    if not rules:
        logger.info("No hay reglas de auto-categorización definidas.")
        result.unmatched = list(transaction_ids)
        return result

    # -- Load transactions ---------------------------------------------------
    transactions = (
        session.query(Transaction)
        .filter(Transaction.id.in_(transaction_ids))
        .all()
    )

    # -- Apply rules to each transaction ------------------------------------
    for txn in transactions:
        _process_transaction(txn, rules, result)

    logger.info(
        "Rules engine completado: %s", result.summary
    )
    return result


def apply_rules_and_commit(
    session:         Session,
    transaction_ids: List[int],
    user_id:         int,
) -> RulesEngineResult:
    """
    Apply rules AND persist the assignments to the DB.
    Conflicted and unmatched transactions are left with category_id=NULL.

    Returns
    -------
    RulesEngineResult (same as apply_rules, with DB writes committed).
    """
    result = apply_rules(session, transaction_ids, user_id)

    # Write assignments
    for txn_id, cat_id in result.assigned.items():
        txn = session.get(Transaction, txn_id)
        if txn is not None:
            txn.category_id = cat_id

    session.commit()
    logger.info(
        "Auto-categorización guardada: %d asignaciones.", len(result.assigned)
    )
    return result


def create_rule(
    session:       Session,
    category_id:   int,
    match_pattern: str,
    match_type:    MatchType  = MatchType.contains,
    priority:      int        = 100,
    user_id:       Optional[int] = None,
) -> AutoCategorizationRule:
    """
    Create and persist a new auto-categorization rule.

    Parameters
    ----------
    session       : Active SQLAlchemy session.
    category_id   : Category to assign when this rule matches.
    match_pattern : Pattern string (substring, exact string, or regex).
    match_type    : contains | exact | regex.
    priority      : Lower = higher priority. Default 100.
    user_id       : None = global rule. User ID = per-user rule.

    Returns
    -------
    AutoCategorizationRule (persisted).

    Raises
    ------
    ValueError : If match_pattern is empty or regex is invalid.
    """
    if not match_pattern or not match_pattern.strip():
        raise ValueError("El patrón de la regla no puede estar vacío.")

    if match_type == MatchType.regex:
        try:
            re.compile(match_pattern)
        except re.error as exc:
            raise ValueError(
                f"Expresión regular inválida '{match_pattern}': {exc}"
            ) from exc

    rule = AutoCategorizationRule(
        user_id       = user_id,
        category_id   = category_id,
        match_pattern = match_pattern.strip(),
        match_type    = match_type,
        priority      = priority,
    )
    session.add(rule)
    session.commit()
    session.refresh(rule)
    logger.info(
        "Regla creada: id=%d pattern='%s' type=%s priority=%d user=%s",
        rule.id, rule.match_pattern, rule.match_type.value,
        rule.priority, user_id,
    )
    return rule

    session: Session,

    Return rules ordered by priority ASC.
    If user_id is provided, returns global rules + rules for that user.

    """

    if user_id is not None:

# Internal helpers

def _load_rules(
    session: Session,
    user_id: int,

    """Load global + user-specific rules, ordered by priority ASC."""
    return get_all_rules(session, user_id=user_id)


def _match(
    rule:  AutoCategorizationRule,
    value: Optional[str],
) -> bool:
    """
    Test a single field value against a rule's pattern.
    Returns True if the pattern matches, False otherwise.
    """
    if not value:
        return False

    pattern = rule.match_pattern
    text    = value.lower()

    if rule.match_type == MatchType.contains:
        return pattern.lower() in text

    elif rule.match_type == MatchType.exact:
        return text == pattern.lower()

    elif rule.match_type == MatchType.regex:
        try:
            return bool(re.search(pattern, value, re.IGNORECASE))
        except re.error:
            logger.warning(
                "Regex inválida en regla id=%d: '%s'", rule.id, pattern
            )
            return False

    return False


def _match_transaction(
    rule: AutoCategorizationRule,
    txn:  Transaction,
) -> Optional[str]:
    """
    Test a rule against a transaction using OR gate on issuer_name + concept.

    Returns
    -------
    str  : 'issuer_name' | 'concept' | 'both' if matched.
    None : if no match.
    """
    hit_issuer  = _match(rule, txn.issuer_name)
    hit_concept = _match(rule, txn.concept)

    if hit_issuer and hit_concept:
        return "both"
    elif hit_issuer:
        return "issuer_name"
    elif hit_concept:
        return "concept"
    return None


def _process_transaction(
    txn:    Transaction,
    rules:  List[AutoCategorizationRule],
    result: RulesEngineResult,
) -> None:
    """
    Apply all rules to a single transaction and update result in-place.

    Logic:
    - Iterate rules in priority order (already sorted ASC).
    - First match at priority P assigns the transaction.
    - If a SECOND rule at the SAME priority P also matches and maps to a
      DIFFERENT category → conflict flagged, no auto-assignment.
    - Once a priority level produces a match (assign or conflict),
      lower-priority rules are ignored.
    """
    best_priority:  Optional[int]  = None
    best_rule:      Optional[AutoCategorizationRule] = None
    best_field:     Optional[str]  = None
    conflict_rules: List[tuple]    = []  # (rule, field_matched)

    for rule in rules:
        field_hit = _match_transaction(rule, txn)
        if field_hit is None:
            continue

        if best_priority is None:
            # First match
            best_priority = rule.priority
            best_rule     = rule
            best_field    = field_hit
            conflict_rules = [(rule, field_hit)]

        elif rule.priority == best_priority:
            # Same priority level — potential conflict
            if rule.category_id != best_rule.category_id:
                conflict_rules.append((rule, field_hit))

        else:
            # Lower priority (higher number) — stop searching
            break

    if not conflict_rules:
        # No match at all
        result.unmatched.append(txn.id)

    elif len(conflict_rules) == 1:
        # Single winner — assign
        result.assigned[txn.id] = best_rule.category_id

    else:
        # Multiple rules at same priority, different categories → conflict
        conflicts = []
        first_rule, first_field = conflict_rules[0]
        for other_rule, other_field in conflict_rules[1:]:
            conflicts.append(RuleConflict(
                transaction_id = txn.id,
                rule_a_id      = first_rule.id,
                rule_b_id      = other_rule.id,
                category_a_id  = first_rule.category_id,

                pattern_b      = other_rule.match_pattern,
            ))
        result.conflicts[txn.id] = conflicts
        logger.warning(
            "Conflicto de reglas para transacción id=%d: "
            "%d reglas coinciden en prioridad=%d",
            txn.id, len(conflict_rules), best_priority,
```

```python
        """Query DB and refresh all KPI cards."""
        today      = date.today()
        date_from, date_to = self._get_date_range(today)

        try:
            # Running balance — latest transaction saldo
            last_txn = (
                self._session.query(Transaction)
                .filter(Transaction.user_id == self._user.id)
                .order_by(Transaction.transaction_date.desc(),
                          Transaction.id.desc())
                .first()
            )
            balance = last_txn.running_balance if last_txn else 0.0

            # Period aggregates
            period_q = (
                self._session.query(
                    Transaction.type,
                    func.sum(Transaction.amount).label("total")
                )
                .filter(
                    Transaction.user_id == self._user.id,
                    Transaction.transaction_date >= date_from,
                    Transaction.transaction_date <= date_to,
                )
                .group_by(Transaction.type)
                .all()
            )

            totals = {row.type: (row.total or 0.0) for row in period_q}

            income     = totals.get(TransactionType.income,     0.0)
            expense    = abs(totals.get(TransactionType.expense, 0.0))
            investment = abs(totals.get(
                TransactionType.investment, 0.0
            ))

            savings_rate = (
                ((income - expense) / income * 100) if income > 0 else 0.0
            )

            # Uncategorized count
            uncat = (
                self._session.query(func.count(Transaction.id))
                .filter(
                    Transaction.user_id    == self._user.id,
                    Transaction.category_id == None,  # noqa: E711
                    Transaction.transaction_date >= date_from,
                    Transaction.transaction_date <= date_to,
                )
                .scalar() or 0
            )

        except Exception:
            balance = expense = income = investment = 0.0
            savings_rate = 0.0
            uncat = 0

        # Update cards
        self._cards["balance"].set_value(
            self._fmt_currency(balance),
            sub=f"A {today.strftime('%d %b %Y')}",
```

```python
def _read_config_value(key: str, fallback: str) -> str:
    """
    Read a string constant from app/config.py using regex.
    Handles both single and double quote assignments.
    Returns `fallback` if the key is not found or file is unreadable.

    config_path = ROOT / "app" / "config.py"

def _escape_fence(code: str, lang: str = "python") -> str:
    """
    Wrap `code` in a markdown fenced block.
    If `code` contains backtick sequences of length >= 3, use a longer fence
    to prevent premature closure (the root cause of Amp upload rejection).
    BUG-EH-05: never emit empty language tag.
    """
    max_run = 0
    run     = 0
    for ch in code:
        if ch == "`":
            run += 1
            max_run = max(max_run, run)
        else:
            run = 0
    fence_len = max(3, max_run + 1)
    fence     = "`" * fence_len
    lang_tag  = lang if lang else "text"
    return "{}{}\n{}\n{}".format(fence, lang_tag, code, fence)

def _write_version_to_config(new_version: str) -> bool:

    Write the chosen snapshot version back to app/config.py.
    Updates APP_VERSION and APP_RELEASE_DATE.
    Refreshes module-level CURRENT_APP_VERSION global after writing.
    """
    global CURRENT_APP_VERSION
    config_path = ROOT / "app" / "config.py"
    if not config_path.exists():
        print("  WARNING: app/config.py not found -- version NOT written back.")
        return False
    try:
        original_text = config_path.read_text(encoding="utf-8")
        lines         = original_text.splitlines(keepends=True)
    except OSError as exc:
        print("  WARNING: Could not read app/config.py: {}".format(exc))
        return False

    new_lines       = []
    version_written = False
    date_written    = False
    now_date        = datetime.now().strftime("%Y-%m-%d")

    for line in lines:
        stripped = line.rstrip("\n\r")

        # Replace APP_VERSION = "..."
        if re.match(r'^APP_VERSION\s*=', stripped) and not version_written:
            eol = "\r\n" if line.endswith("\r\n") else "\n"
            new_lines.append('APP_VERSION = "{}"{}'.format(new_version, eol))
            version_written = True

        # Replace APP_RELEASE_DATE = "..."
        elif re.match(r'^APP_RELEASE_DATE\s*=', stripped) and not date_written:
            eol = "\r\n" if line.endswith("\r\n") else "\n"
            new_lines.append(
                'APP_RELEASE_DATE = "{}"{}'.format(now_date, eol))
            date_written = True

        else:
            new_lines.append(line)

    if not version_written:
        print("  WARNING: APP_VERSION line not found in app/config.py"
              " -- not updated.")
        return False

    try:
        config_path.write_text(
            "".join(new_lines), encoding="utf-8", newline="\n")
    except OSError as exc:
        print("  WARNING: Could not write app/config.py: {}".format(exc))
        return False

    # Refresh global so the rest of this run uses the new version
    CURRENT_APP_VERSION = new_version
    print('  app/config.py updated -> APP_VERSION = "{}"'.format(new_version))
    if date_written:
        print('  app/config.py updated -> APP_RELEASE_DATE = "{}"'.format(
            now_date))
    else:
        print("  app/config.py: APP_RELEASE_DATE not found -- "
              "add it to app/config.py if needed.")
    return True

def _write_prj_archive(
    prj_path:   Path,
    file_list:  List[str],
    manifest:   Dict,
) -> None:
    """
    Write a .prj portable archive (ZIP) containing all source files
    and a manifest.json with snapshot metadata.
    """
    with zipfile.ZipFile(prj_path, "w", zipfile.ZIP_DEFLATED) as zf:
        # Write manifest
        zf.writestr(
            "manifest.json",
            json.dumps(manifest, indent=2, ensure_ascii=False),
        )
        # Write source files
        for rel in file_list:
            src = ROOT / rel
            if src.exists():
                try:
                    zf.write(src, arcname="source/{}".format(rel))
                except Exception as exc:
                    print("  WARNING: Could not archive {}: {}".format(
                        rel, exc))


def _read_prj_manifest(prj_path: Path) -> Optional[Dict]:
    """Read and return the manifest.json from a .prj archive."""
    try:
        with zipfile.ZipFile(prj_path, "r") as zf:
            if "manifest.json" in zf.namelist():
                return json.loads(
                    zf.read("manifest.json").decode("utf-8", errors="replace"))
    except Exception as exc:
        print("  ERROR: Cannot read .prj archive {}: {}".format(
            prj_path.name, exc))
    return None

def export_version(
    version:     Optional[str] = None,
    ver_type:    Optional[str] = None,
    output_path: Optional[Path] = None,
    also_prj:    Optional[bool] = None,
) -> None:
    """
    Write all project source files to a versioned markdown snapshot
    stored in ROOT/versions/.

    Parameters
    ----------
    version     : Snapshot label e.g. "1.0". None -> interactive prompt.
    ver_type    : "B" (Beta) or "R" (Release). None -> interactive prompt.
    output_path : Override output .md path (optional, Path object).
    also_prj    : If True, also write a .prj archive. None -> prompt user.
    """
    _ensure_dir(VERSIONS_DIR)

    # -- Version type prompt --------------------------------------------------
    if ver_type is None:
        ver_type = _prompt_version_type()

    # -- Version number prompt ------------------------------------------------
    if version is None:
        version, is_override = _prompt_export_version(ver_type)
    else:
        is_override = False

    # -- Description (new revisions only) -------------------------------------
    description = ""
    if not is_override:
        description = _prompt_revision_description()

    # -- .prj prompt ----------------------------------------------------------
    if also_prj is None:
        also_prj = _prompt_also_prj()

    # -- Build output paths ---------------------------------------------------
    snap_base   = _snapshot_filename(CURRENT_APP_NAME, ver_type, version)
    if output_path is None:
        output_path = VERSIONS_DIR / "{}.md".format(snap_base)
    prj_path = VERSIONS_DIR / "{}.prj".format(snap_base) if also_prj else None

    # -- Overwrite guard ------------------------------------------------------
    if not _check_existing_file(output_path):
        return

    # -- Update version in app/config.py (new revisions only) ----------------
    if not is_override:
        _write_version_to_config(version)

    # -- Build file list ------------------------------------------------------
    ordered, auto_added = _build_file_list()
    now = datetime.now().strftime("%Y-%m-%d %H:%M")

    # -- Read & embed files ---------------------------------------------------
    sections = []
    missing  = []
    present  = []

    for rel in ordered:
        src = ROOT / rel
        if not src.exists():
            missing.append(rel)
            continue
        present.append(rel)
        try:
            code = src.read_text(encoding="utf-8", errors="backslashreplace")
        except Exception as exc:
            code = "# ERROR reading file: {}\n".format(exc)

        ext      = src.suffix.lstrip(".").lower() or "text"
        lang_map = {
            "py":   "python",
            "md":   "markdown",
            "ini":  "ini",
            "cfg":  "ini",
            "toml": "toml",
            "txt":  "text",
            "json": "json",
            "html": "html",
            "bat":  "batch",
            "sh":   "bash",
        }
        lang = lang_map.get(ext, "text")

        # BUG-EH-03 FIX: no trailing space before \n in heading
        sections.append(
            "## `{}`\n\n{}\n\n".format(rel, _escape_fence(code, lang))
        )

    file_count = len(present)

    # -- Table of Contents ----------------------------------------------------
    toc = "\n".join(
        "- [`{}`](#{})".format(
            rel,
            rel.lower()
               .replace("/", "")
               .replace(".", "")
               .replace("_", "-")
               .replace(" ", "-"),
        )
        for rel in present
    )

    # -- Build header notes ---------------------------------------------------
    override_note    = (
        "> **⚠ Override export** — same version re-exported.  \n"
        if is_override else ""
    )
    description_note = (
        "> **Description:** {}  \n".format(description)
        if description else ""
    )
    auto_note = (
        "> **Auto-discovered:** {} extra file(s) appended.  \n".format(
            len([f for f in auto_added if (ROOT / f).exists()]))
        if auto_added else ""
    )

    # -- Assemble header ------------------------------------------------------
    header = (
        "# {} -- Full Python Source Export\n\n"
        "> **Version type:** {} ({})\n"
        "> **Version:** {}  \n"
        "> **App name  (app/config.py):** {}  \n"
        "> **App version (app/config.py):** {}  \n"
        "> **Export date:** {}  \n"
        "> **Files included:** {}  \n"
        "> **Handler version:** {}  \n"
        "> **Generated by:** `tools/ExportVersionHandler.py`  \n"
        "{}"
        "{}"
        "{}"
        "\n---\n\n"
        "## Table of Contents\n\n{}\n\n---\n\n"
    ).format(
        snap_base,
        "Beta" if ver_type == "B" else "Release", ver_type,
        version,
        CURRENT_APP_NAME,
        CURRENT_APP_VERSION,
        now,
        file_count,
        HANDLER_VERSION,
        override_note,
        description_note,
        auto_note,
        toc,
    )

    if missing:
        header += (
            "## WARNING: Missing Files (skipped)\n\n"
            + "\n".join("- `{}`".format(f) for f in missing)
            + "\n\n---\n\n"
        )
    if auto_added:
        present_extras = [f for f in auto_added if (ROOT / f).exists()]
        header += (
            "## Auto-Discovered Files (not in PRIORITY_FILES)\n\n"
            + "\n".join("- `{}`".format(f) for f in present_extras)
            + "\n\n---\n\n"
        )

    # -- Write .md snapshot ---------------------------------------------------
    full_content = header + "".join(sections)
    try:
        output_path.write_text(full_content, encoding="utf-8", newline="\n")
    except OSError as exc:
        print("  ERROR: Could not write snapshot: {}".format(exc))
        return

    size_kb = output_path.stat().st_size / 1024

    # -- Write .prj archive ---------------------------------------------------
    if also_prj and prj_path:
        manifest = {
            "snap_base":    snap_base,
            "ver_type":     ver_type,
            "app_name":     CURRENT_APP_NAME,
            "app_version":  CURRENT_APP_VERSION,
            "export_date":  now,
            "handler":      HANDLER_VERSION,
            "description":  description,
            "is_override":  is_override,
            "files":        present,
        }
        try:
            _write_prj_archive(prj_path, present, manifest)
        except Exception as exc:
            print("  ERROR: Could not write .prj archive: {}".format(exc))

    # -- Update CHANGELOG (new revisions only) --------------------------------
    if not is_override:
        _update_changelog(ver_type, version, now, file_count,
                          description, snap_base)

    # -- Final summary --------------------------------------------------------
    print("\n  {}".format(_sep("=")))
    print("  EXPORT COMPLETE")
    print("  {}".format(_sep("-")))
    print("  Snapshot : {}".format(snap_base))
    print("  Type     : {}".format("Beta" if ver_type == "B" else "Release"))
    print("  Files    : {}  ({} missing, {} auto-added)".format(
        file_count, len(missing), len(auto_added)))
    print("  .md      : {}  ({:.1f} KB)".format(output_path.name, size_kb))
    if prj_path and prj_path.exists():
        prj_kb = prj_path.stat().st_size / 1024
        print("  .prj     : {}  ({:.1f} KB)".format(prj_path.name, prj_kb))
    if not is_override:
        print("  Desc.    : {}".format(description))
        print("  CHANGELOG: CHANGELOG.md updated")
    else:
        print("  Override : CHANGELOG.md NOT modified")
    print("  {}\n".format(_sep("=")))

def import_snapshot() -> None:
    """

    """

    print("  SELECT SNAPSHOT TO INSPECT")
    print(_sep())
    for i, f in enumerate(existing, 1):
        print("  [{:2d}]  {}".format(i, f.name))

        if manifest is None:
            print("  ERROR: Cannot read manifest.")

            print("  Desc.     : {}".format(desc))

        for rel in manifest.get("files", []):
            print("    {}".format(rel))

        print("\n  Extract to staging folder for review? [y/n]: ", end="")

                with zipfile.ZipFile(chosen, "r") as zf:
                    zf.extractall(staging)
                print("  Extracted to: {}".format(
                    staging.relative_to(ROOT)))
            except Exception as exc:
                print("  ERROR: Extraction failed: {}".format(exc))
    else:
        # .md file — show first 40 lines (header block)
        try:
            lines = chosen.read_text(
                encoding="utf-8", errors="replace").splitlines()
        except OSError as exc:
            print("  ERROR: Cannot read file: {}".format(exc))

        print("  {}".format(_sep("=")))

def restore_snapshot() -> None:
    """Interactive restore from a .prj archive with full backup."""
    _ensure_dir(VERSIONS_DIR)
    existing  = _list_existing_versions()
    prj_files = [f for f in existing if f.suffix == ".prj"]
    if not prj_files:
        print("\n  No .prj archives found in versions/ -- nothing to restore.")
        return

    print("\n  {}".format(_sep()))
    print("  SELECT .prj ARCHIVE TO RESTORE")
    print(_sep())
    for i, f in enumerate(prj_files, 1):
        print("  [{:2d}]  {}".format(i, f.name))
    print("  [ 0]  Cancel")
    print(_sep())

    while True:
        raw = input("\n  Enter number: ").strip()
        if raw == "0":
            print("  Restore cancelled.")
            return
        try:
            idx = int(raw) - 1
            if 0 <= idx < len(prj_files):
                chosen = prj_files[idx]
                break
        except ValueError:
            pass
        print("  Please enter a number between 0 and {}.".format(
            len(prj_files)))

    manifest = _read_prj_manifest(chosen)
    if manifest is None:
        print("  ERROR: Cannot read manifest -- restore aborted.")
        return

    files_to_restore = manifest.get("files",       [])
    snap_base        = manifest.get("snap_base",    "unknown")
    snap_type        = manifest.get("ver_type",     "?")
    snap_app_ver     = manifest.get("app_version",  "unknown")
    snap_date        = manifest.get("export_date",  "unknown")
    snap_desc        = manifest.get("description",  "")

    print("\n  {}".format(_sep("=")))
    print("  Snapshot  : {}".format(snap_base))
    print("  Type      : {}".format(
        "Beta" if snap_type == "B" else "Release"))
    print("  App ver   : {}".format(snap_app_ver))
    print("  Exported  : {}".format(snap_date))
    print("  Files     : {}".format(len(files_to_restore)))
    if snap_desc:
        print("  Desc.     : {}".format(snap_desc))
    print("  {}".format(_sep("-")))

    will_overwrite = []
    will_create    = []
    for rel in files_to_restore:
        if (ROOT / rel).exists():
            will_overwrite.append(rel)
            print("  [OVERWRITE]  {}".format(rel))
        else:
            will_create.append(rel)
            print("  [NEW FILE]   {}".format(rel))

    if will_overwrite:
        print("\n  WARNING: {} file(s) will be OVERWRITTEN.".format(
            len(will_overwrite)))
    if will_create:
        print("  {} new file(s) will be CREATED.".format(len(will_create)))
    print("  All existing files will be backed up to versions/restores/")
    print("  {}".format(_sep("=")))
    print("\n  THIS WILL OVERWRITE LIVE PROJECT FILES.")
    print("  Type  CONFIRM  to proceed, or press Enter to cancel safely.")
    if input("\n  Type CONFIRM to restore: ").strip() != "CONFIRM":
        print("  Restore cancelled -- live project unchanged.")
        return

    # -- Perform restore with backup ------------------------------------------
    backup_ts  = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_dir = RESTORE_DIR / "backup_{}".format(backup_ts)
    _ensure_dir(backup_dir)

    restored = 0
    errors   = 0

    try:
        with zipfile.ZipFile(chosen, "r") as zf:
            for rel in files_to_restore:
                dest = ROOT / rel

                # Backup existing file first
                if dest.exists():
                    bak = backup_dir / rel
                    bak.parent.mkdir(parents=True, exist_ok=True)
                    try:
                        shutil.copy2(dest, bak)
                    except Exception as exc:
                        print("  WARNING: Backup failed for {}: {}".format(
                            rel, exc))

                # Write restored file
                src_key = "source/{}".format(rel)
                if src_key not in zf.namelist():
                    print("  WARNING: Not found in archive: {}".format(rel))
                    errors += 1
                    continue
                dest.parent.mkdir(parents=True, exist_ok=True)
                try:
                    dest.write_bytes(zf.read(src_key))
                    print("  RESTORED: {}".format(rel))
                    restored += 1
                except Exception as exc:
                    print("  ERROR: {}: {}".format(rel, exc))
                    errors += 1
    except Exception as exc:
        print("  ERROR: Restore failed: {}".format(exc))
        return

    # -- Write restore manifest -----------------------------------------------
    restore_manifest = {
        "restored_from":  chosen.name,
        "backup_dir":     str(backup_dir.relative_to(ROOT)),
        "restored_at":    backup_ts,
        "files_restored": files_to_restore,
        "files_created":  will_create,
    }
    try:
        (backup_dir / "restore_manifest.json").write_text(
            json.dumps(restore_manifest, indent=2, ensure_ascii=False),
            encoding="utf-8",
            newline="\n",
        )
    except Exception:
        pass

    print("\n  {}".format(_sep("=")))
    print("  RESTORE COMPLETE")
    print("  Restored: {}  |  Errors: {}".format(restored, errors))
    print("  Backup  : {}".format(backup_dir.relative_to(ROOT)))
    print("  {}\n".format(_sep("=")))

def undo_restore() -> None:
    """Revert to the most recent backup set created by restore_snapshot()."""
    if not RESTORE_DIR.exists():
        print("\n  No restores directory found -- nothing to undo.")
        return
    backups = sorted(
        [b for b in RESTORE_DIR.iterdir() if b.is_dir()], reverse=True)
    if not backups:
        print("\n  No backup sets found in versions/restores/ "
              "-- nothing to undo.")
        return

    latest = backups[0]
    print("\n  Most recent backup: {}".format(latest.name))
    manifest_path = latest / "restore_manifest.json"
    if not manifest_path.exists():
        print("  WARNING: No restore_manifest.json found "
              "-- cannot determine files to undo.")
        return
    try:
        restore_manifest = json.loads(
            manifest_path.read_text(encoding="utf-8"))
    except Exception as exc:
        print("  ERROR: Cannot read restore manifest: {}".format(exc))
        return

    files_to_undo = restore_manifest.get("files_restored", [])
    files_created = restore_manifest.get("files_created",  [])
    print("  Files to revert : {}".format(len(files_to_undo)))
    print("  Files to remove : {}".format(len(files_created)))
    confirm = input("\n  Undo this restore? [y/n]: ").strip().lower()
    if confirm != "y":
        print("  Undo cancelled.")
        return

    undone  = 0
    removed = 0
    errors  = 0

    for rel in files_to_undo:
        src  = latest / rel
        dest = ROOT / rel
        if not src.exists():
            print("  WARNING: Backup not found: {}".format(rel))
            errors += 1
            continue
        dest.parent.mkdir(parents=True, exist_ok=True)
        try:
            shutil.copy2(src, dest)
            print("  REVERTED: {}".format(rel))
            undone += 1
        except Exception as exc:
            print("  ERROR: {}: {}".format(rel, exc))
            errors += 1

    for rel in files_created:
        dest = ROOT / rel
        if dest.exists():
            try:
                dest.unlink()
                print("  REMOVED: {}".format(rel))
                removed += 1
            except Exception as exc:
                print("  WARNING: Could not remove {}: {}".format(rel, exc))

    print("\n  {}".format(_sep("=")))
    print("  UNDO COMPLETE")
    print("  Reverted: {}  |  Removed: {}  |  Errors: {}".format(
        undone, removed, errors))
    print("  {}\n".format(_sep("=")))
```

```python
def _hash_password(plain: str) -> str:
    """
    Hash a plaintext password using bcrypt (rounds=12).
    Returns a UTF-8 decoded hash string for storage in the DB.
    """
    return bcrypt.hashpw(
        plain.encode("utf-8"),
        bcrypt.gensalt(rounds=12),
    ).decode("utf-8")


def _verify_password(plain: str, hashed: str) -> bool:
    """
    Verify a plaintext password against a stored bcrypt hash.
    Returns True if the password matches, False otherwise.
    Safe against all bcrypt exceptions.
    """
    try:
        return bcrypt.checkpw(
            plain.encode("utf-8"),
            hashed.encode("utf-8"),
        )
    except Exception:
        return False

    ----------
    session  : Active SQLAlchemy session.
    username : Must be non-empty string. Uniqueness enforced at DB level.
    password : Must be non-empty string. Stored as bcrypt hash only.

    Returns
    -------
    User ORM object (already committed to the session).

    Raises
    ------
    ValueError : If username or password is empty.
    ValueError : If username already exists (duplicate).
    """
    if not username or not username.strip():
        raise ValueError("El nombre de usuario no puede estar vacío.")
    if not password:
        raise ValueError("La contraseña no puede estar vacía.")

    username      = username.strip()
    password_hash = _hash_password(password)

    user = User(
        username      = username,
        password_hash = password_hash,
        created_at    = datetime.now(timezone.utc),
    )
    session.add(user)

    try:
        session.commit()
        session.refresh(user)
        logger.info("Usuario creado: '%s' (id=%s)", username, user.id)
        return user

    except IntegrityError:
        session.rollback()
        raise ValueError(
            f"El usuario '{username}' ya existe. "
            "Elige un nombre de usuario diferente."
        )


def verify_login(
    session:  Session,
    username: str,
    password: str,
) -> Optional[User]:
    """
    Verify login credentials and update last_login on success.

    Returns
    -------
    User object if credentials are valid, None otherwise.
    """
    if not username or not password:
        return None

    user: Optional[User] = (
        session.query(User)
        .filter(User.username == username.strip())
        .first()
    )

    if user is None:
        logger.debug("Login fallido: usuario '%s' no encontrado.", username)
        return None

    if not _verify_password(password, user.password_hash):
        logger.debug("Login fallido: contraseña incorrecta para '%s'.", username)
        return None

    # Update last_login
    user.last_login = datetime.now(timezone.utc)
    try:
        session.commit()
    except Exception as exc:
        session.rollback()
        logger.warning(
            "No se pudo actualizar last_login para '%s': %s", username, exc
        )

    logger.info("Login exitoso: '%s' (id=%s)", username, user.id)
    return user


def get_user_by_id(session: Session, user_id: int) -> Optional[User]:
    """Retrieve a user by primary key. Returns None if not found."""
    return session.get(User, user_id)
```

```python
"""
In-process session manager for PROFIN.

Holds the currently authenticated user for the lifetime of the application.
A new login clears any previous session. Session is cleared on app exit.

This is intentionally simple (single-user, single-process) — no tokens,
no persistence between restarts.
"""

from __future__ import annotations

from typing import Optional
from app.db.models import User


class SessionManager:
    """Singleton-style in-memory session store."""

    _current_user: Optional[User] = None

    @classmethod
    def login(cls, user: User) -> None:
        """Set the active session user."""
        cls._current_user = user

    @classmethod
    def logout(cls) -> None:
        """Clear the active session."""
        cls._current_user = None

    @classmethod
    def get_current_user(cls) -> Optional[User]:
        """Return the currently logged-in user, or None."""
        return cls._current_user

    @classmethod
    def is_authenticated(cls) -> bool:
        """Return True if a user is currently logged in."""
        return cls._current_user is not None

    @classmethod
    def require_user(cls) -> User:
        """
        Return the current user or raise RuntimeError.
        Use in GUI handlers where a logged-in user is always expected.
        """
        if cls._current_user is None:
            raise RuntimeError(
                "No hay ningún usuario autenticado. "
                "Por favor, inicia sesión antes de continuar."
            )
        return cls._current_user
```
